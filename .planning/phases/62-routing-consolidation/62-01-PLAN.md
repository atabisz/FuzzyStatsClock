  Add this method near the other locale helper methods (after SetLanguage, around line 1441). Exact implementation:

  ```csharp
  private string ResolveLocaleKey(string phraseLocale, string phraseStyle, string uiLang)
  {
      string styleLower = phraseStyle.ToLowerInvariant();

      if (phraseLocale is "fr" or "es" or "de" or "pl")
          return phraseLocale;

      if (phraseLocale == "ja")
          return styleLower switch
          {
              "terse"  => "ja-terse",
              "poetic" => "ja-poetic",
              "rude"   => "ja-rude",
              _        => "ja-classic",
          };

      if (phraseLocale == "en")
          return styleLower switch
          {
              "terse"       => "en-terse",
              "poetic"      => "en-poetic",
              "rude"        => "en-rude",
              "pirate"      => "en-pirate",
              "dwarf"       => "en-dwarf",
              "jive"        => "en-jive",
              "valleygirl"  => "en-valleygirl",
              "yoda"        => "en-yoda",
              "shakespeare" => "en-shakespeare",
              _             => "en-classic",
          };

      // "auto" — detect from Windows UI culture
      if (uiLang is "fr" or "es" or "de" or "pl")
          return uiLang;
      if (uiLang == "ja")
          return styleLower switch
          {
              "terse"  => "ja-terse",
              "poetic" => "ja-poetic",
              "rude"   => "ja-rude",
              _        => "ja-classic",
          };
      // auto + English system
      return styleLower switch
      {
          "terse"       => "en-terse",
          "poetic"      => "en-poetic",
          "rude"        => "en-rude",
          "pirate"      => "en-pirate",
          "dwarf"       => "en-dwarf",
          "jive"        => "en-jive",
          "valleygirl"  => "en-valleygirl",
          "yoda"        => "en-yoda",
          "shakespeare" => "en-shakespeare",
          _             => "en-classic",
      };
  }
  ```
